/-- The fragment: every column of height ≤ 2 with row-1 entry ≤ 2. -/
def inFragC (m : BMS.Matrix) : Bool := m.all fun c => c.length ≤ 2 && r1 c ≤ 2

/-- Is `t` a strict fixed point of φ_k (its head's φ-level lies above k)? -/
def isHighFp (k : Term) : Term → Bool
  | .add a _ => isHighFp k a
  | .phi a _ => lt k a
  | _ => false

/-- The additive delta of `v` above the thread `thr`: 0 if v = thr, the
    remainder if thr is an additive prefix of v, v itself otherwise. -/
def stripThr (thr v : Term) : Term :=
  if thr == Term.zero then v
  else
    let lt := toList thr
    let lv := toList v
    if lv.take lt.length == lt then ofList (lv.drop lt.length) else v

mutual

/-- The K-arithmetic reading (see the header).  `kb` is the enclosing
    (0,1)-nesting level, `lacc0` the inherited level-space accumulator;
    the fold carries (acc, lacc, aacc). -/
def oCAux : Nat → Term → Term → Term → List BMS.Col → Term
  | 0, _, _, _, _ => Term.zero
  | fuel + 1, kb, lacc0, acc0, s =>
    ((blocksP s).foldl (init := (acc0, lacc0, Term.zero)) fun (acc, lacc, aacc) b =>
      match b with
      | [] => (acc, lacc, aacc)
      | c :: t =>
        if r1 c == 0 then
          -- summand rule; inside a (0,2)-context it runs in the argument space
          let x := omegaNF (oCAux fuel Term.zero Term.zero Term.zero (decP t))
          if lacc == Term.zero then
            (plus acc x, lacc, aacc)
          else
            let k := plus kb (omegaNF lacc)
            let aacc' := plus aacc x
            (match logPhi k acc with
             | none => plus acc x
             | some bse => phiNF k (plus bse (omegaNF aacc')),
             lacc, aacc')
        else if r1 c == 1 then
          -- (0,1)-block: successor-level φ-step one level above the context
          let k1 := plus (if lacc == Term.zero then kb
                          else plus kb (omegaNF lacc)) one
          let thr := if isHighFp (plus k1 one) acc then acc else Term.zero
          let vi := oCAux fuel k1 Term.zero thr (decP t)
          (phiStep k1 acc (stripThr thr vi), lacc, Term.zero)
        else
          -- (0,2)-block: the ψ-branch at the Ω-floor when the level material
          -- opens at row-1 ≥ 1 (the level climb hits Ω, header §3); the
          -- level-space climb otherwise
          let L := decP t
          let climb := fun (_ : Unit) =>
            let lacc' := plus lacc
              (omegaNF (oCAux fuel kb (plus lacc one) Term.zero L))
            let k2 := plus kb (omegaNF lacc')
            let g := match logPhi k2 acc with
                     | none => Term.zero
                     | some bse => plus bse (omegaNF aacc)
            (phiNF k2 g, lacc', Term.zero)
          if kb == one && lacc == Term.zero &&
             (match L.head? with | some ch => r1 ch ≥ 1 | none => false) then
            match blocksP L with
            | [c1 :: amat] =>
              if r1 c1 == 1 then
                -- the ψ-configuration: collapse of the argument read
                (plus acc (psi Om (oCArg fuel (decP amat))), lacc, aacc)
              else if amat.isEmpty then
                -- the (0,2)-diagonal (2,2)(3,2) = ψ_Ω(ω̄^(Ω+2))
                (plus acc (psi Om (omegaNF (plus Om (ofNat 2)))), lacc, aacc)
              else climb ()   -- deeper (3,2)-shapes: frontier (header §4)
            | _ => climb ()   -- after-signal residue material: frontier (§4)
          else climb ()).1

/-- The ψ-argument read (header §3): value of the ψ-signal block's dec'd
    tail.  A single block headed at row-1 ≥ 1 stays unwrapped (the Ω-scale
    sums); everything else is ω̄-wrapped via `omegaNF` (the sub-Ω exponents
    and the Ω-power diagonals). -/
def oCArg : Nat → List BMS.Col → Term
  | 0, _ => Term.zero
  | fuel + 1, mat =>
    match blocksP mat with
    | [] => Term.zero
    | [b] =>
      (match b.head? with
       | some c => if r1 c ≥ 1 then oCArgB fuel true b
                   else omegaNF (oCArgB fuel false b)
       | none => Term.zero)
    | bs => omegaNF (bs.foldl (fun a b => plus a (oCArgB fuel false b)) Term.zero)

/-- One argument-space block (header §3):
    (0,0)-block — ω̄^(fresh read of the dec'd tail);
    (0,1)-block — Ω plus the flat read of its material, ω̄-wrapped when the
      material opens at row-1 = 2;
    (0,2)-block — ω̄^(Ω+1) plus the flat material at the top level,
      1 plus the flat material when nested. -/
def oCArgB : Nat → Bool → List BMS.Col → Term
  | 0, _, _ => Term.zero
  | fuel + 1, top, b =>
    match b with
    | [] => Term.zero
    | c :: bt =>
      if r1 c == 0 then
        omegaNF (oCAux fuel Term.zero Term.zero Term.zero (decP bt))
      else if r1 c == 1 then
        let dm := decP bt
        match dm.head? with
        | some ch =>
          if r1 ch == 2 then omegaNF (plus Om (oCArgS fuel dm))
          else plus Om (oCArgS fuel dm)
        | none => Om
      else
        let inner := oCArgS fuel (decP bt)
        if top then plus (omegaNF (plus Om one)) inner else plus one inner

/-- Flat (nested) sum of argument-space blocks. -/
def oCArgS : Nat → List BMS.Col → Term
  | 0, _ => Term.zero
  | fuel + 1, mat =>
    (blocksP mat).foldl (fun a b => plus a (oCArgB fuel false b)) Term.zero

end

end StageC

/-- Stage C (C'''' revision): the translation on the 2-row fragment with
    row-1 entries ≤ 2 (`none` outside).  Row-1-all-zero matrices delegate
    to `oPr`.  The residue-frontier configurations of header §4 are
    provisional. -/
def oStageC? (m : BMS.Matrix) : Option Term :=
  if StageC.inFragC m then
    some (if onlyRow0 m then oPr m
          else StageC.oCAux (4 * m.length + 8) Term.zero Term.zero Term.zero m)
  else none

/-- Total version for the corpus checkers (junk 0 outside the fragment). -/
def oStageCT (m : BMS.Matrix) : Term := (oStageC? m).getD Term.zero
